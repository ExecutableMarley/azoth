103inline std::ostream& operator<<(std::ostream& os, const ProcessImage& img)
104{
105 if (!img.valid())
106 {
107 return os << "ProcessImage{ invalid }";
108 }
109
110 std::ios_base::fmtflags f(os.flags());
111 const char oldFill = os.fill();
112
113 os << "ProcessImage{ "
114 << "baseAddress= " << img.baseAddress
115 << ", size=" << std::dec << img.size;
116
117 if (!img.name.empty())
118 os << ", name=\"" << img.name << "\"";
119
120 if (!img.path.empty())
121 os << ", path=\"" << img.path << "\"";
122
123 os << " }";
124
125 os.flags(f);
126 os.fill(oldFill);
127 return os;
128}
