174class ProtectionFilter
175{
176public:
180 enum ProtectionState : uint8_t
181 {
182 Ignore,
183 Require,
184 Forbid
185 };
186
188 ProtectionState read = ProtectionState::Ignore;
190 ProtectionState write = ProtectionState::Ignore;
192 ProtectionState execute = ProtectionState::Ignore;
193
195 constexpr ProtectionFilter() = default;
196
198 constexpr ProtectionFilter(ProtectionState r, ProtectionState w, ProtectionState e)
199 : read(r), write(w), execute(e) {}
200
206 static constexpr ProtectionFilter exact(EMemoryProtection prot)
207 {
208 return {
209 (prot & EMemoryProtection::Read) != EMemoryProtection::None ? ProtectionState::Require : ProtectionState::Forbid,
210 (prot & EMemoryProtection::Write) != EMemoryProtection::None ? ProtectionState::Require : ProtectionState::Forbid,
211 (prot & EMemoryProtection::Execute) != EMemoryProtection::None ? ProtectionState::Require : ProtectionState::Forbid
212 };
213 }
214
220 static constexpr ProtectionFilter require(EMemoryProtection prot)
221 {
222 return {
223 (prot & EMemoryProtection::Read) != EMemoryProtection::None ? ProtectionState::Require : ProtectionState::Ignore,
224 (prot & EMemoryProtection::Write) != EMemoryProtection::None ? ProtectionState::Require : ProtectionState::Ignore,
225 (prot & EMemoryProtection::Execute) != EMemoryProtection::None ? ProtectionState::Require : ProtectionState::Ignore
226 };
227 }
228
240 static constexpr ProtectionFilter mask(int8_t r, int8_t w, int8_t e)
241 {
242 auto conv = [](int8_t v) constexpr -> ProtectionState {
243 return (v > 0) ? ProtectionState::Require :
244 (v < 0) ? ProtectionState::Forbid :
245 ProtectionState::Ignore;
246 };
247 return { conv(r), conv(w), conv(e) };
248 }
249
251 constexpr ProtectionFilter& requireRead() { read = ProtectionState::Require; return *this; }
253 constexpr ProtectionFilter& forbidRead() { read = ProtectionState::Forbid; return *this; }
255 constexpr ProtectionFilter& requireWrite() { write = ProtectionState::Require; return *this; }
257 constexpr ProtectionFilter& forbidWrite() { write = ProtectionState::Forbid; return *this; }
259 constexpr ProtectionFilter& requireExec() { execute = ProtectionState::Require; return *this; }
261 constexpr ProtectionFilter& forbidExec() { execute = ProtectionState::Forbid; return *this; }
262
266 constexpr bool checkProtection(ProtectionState filterState, EMemoryProtection actual, EMemoryProtection flag) const
267 {
268 bool isPresent = (actual & flag) != EMemoryProtection::None;
269
270 if (filterState == ProtectionState::Require)
271 {
272 return isPresent;
273 }
274 else if (filterState == ProtectionState::Forbid)
275 {
276 return !isPresent;
277 }
278 return true;
279 }
280
284 constexpr bool matchesProtection(EMemoryProtection actual) const
285 {
286 if (!checkProtection(read, actual, EMemoryProtection::Read))
287 {
288 return false;
289 }
290
291 if (!checkProtection(write, actual, EMemoryProtection::Write))
292 {
293 return false;
294 }
295
296 if(!checkProtection(execute, actual, EMemoryProtection::Execute))
297 {
298 return false;
299 }
300
301 return true;
302 }
303
310 constexpr bool operator()(const MemoryRegion& r) const
311 {
312 return matchesProtection(r.protection);
313 }
314};
