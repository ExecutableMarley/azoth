123 bool handleFailure(EPlatformError platformError, uint64_t osError = 0, bool exitCheck = false) const
124 {
125 if (platformError == EPlatformError::Success) return setError(platformError, osError);
126
127 if (exitCheck && !this->_isDead)
128 {
129 auto now = std::chrono::steady_clock::now();
130 if ((now - _lastAliveCheck) >= ALIVE_CHECK_INTERVAL)
131 {
132 _lastAliveCheck = now;
133 bool isAliveResult;
134 if (isAlive(isAliveResult) && !isAliveResult)
135 {
136 _isDead = true;
137 }
138 }
139 }
140 return setError(platformError, osError);
141 }
142
143 bool handleFailure(PlatformErrorState state, bool exitCheck = false) const
